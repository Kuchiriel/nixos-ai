## Build Vulkan (se precisar recompilar)
```
nix shell nixpkgs#cmake nixpkgs#gcc nixpkgs#gnumake nixpkgs#pkg-config nixpkgs#zlib \
  nixpkgs#vulkan-headers nixpkgs#vulkan-loader nixpkgs#shaderc nixpkgs#glslang \
  nixpkgs#spirv-headers --command bash -c '
L=$(nix path-info nixpkgs#vulkan-loader); H=$(nix path-info nixpkgs#vulkan-headers)
SP=$(nix path-info nixpkgs#spirv-headers); export CMAKE_PREFIX_PATH="$L;$H;$SP"
cmake -S . -B build-vk -DCMAKE_BUILD_TYPE=Release -DSD_VULKAN=ON \
  -DVulkan_LIBRARY=$L/lib/libvulkan.so -DVulkan_INCLUDE_DIR=$H/include \
  -DCMAKE_CXX_FLAGS="-I$SP/include"'
```
